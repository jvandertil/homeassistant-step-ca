#include <tunables/global>

profile step-ca-client flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Runtime files are read-only; writable state is listed below.
  /init r,
  /bin/** r,
  /command/** r,
  /etc/** r,
  /lib/** r,
  /package/** r,
  /sbin/** r,
  /usr/** r,
  /var/** r,

  network inet stream,
  network inet6 stream,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** ix,
  /etc/services.d/** ix,
  /etc/cont-init.d/** ix,
  /etc/cont-finish.d/** ix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Configuration, MQTT topic state, and renewal timestamps.
  /data/** rw,

  # Configured certificate filenames are validated as /ssl basenames. The
  # workers also stage and preserve pairs in these fixed recovery directories.
  /ssl/ r,
  /ssl/* rw,
  /ssl/.step-ca-{server,client}-recovery/ rw,
  /ssl/.step-ca-{server,client}-recovery/** rw,

  # Start new profile for service
  /usr/bin/step cx -> step,

  profile step flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_step-ca-client,

    # Each certificate profile supplies its own STEPPATH.
    /root/.step-server/ rw,
    /root/.step-server/** rw,
    /root/.step-client/ rw,
    /root/.step-client/** rw,

    # CA roots and certificate material have configured /ssl basenames;
    # issuance output is staged in a fixed recovery directory.
    /ssl/ r,
    /ssl/* rw,
    /ssl/.step-ca-{server,client}-recovery/ rw,
    /ssl/.step-ca-{server,client}-recovery/** rw,

    network inet stream,
    network inet6 stream,

    # step-cli is called directly by the certificate workers.
    /usr/bin/step mr,
    /etc/passwd r,

    # Allow read cpuset
    /proc/*/cpuset r,

    # Access network
    /sys/devices/virtual/dmi/id/product_uuid r,
    /etc/resolv.conf r,
    /etc/nsswitch.conf r,
    /etc/hosts r,
  }
}
